\section{经典场论}
\subsection{拉氏量、作用量与Euler-Lagrange方程}
\begin{definition}[拉氏量\&拉氏量密度]
    拉氏量
    \begin{equation}
        L(t)=\int \mathrm d^3x \, \mathcal{L}(\phi, \partial_\mu \phi)
    \end{equation}
    其中$\mathcal L$即拉氏量密度
\end{definition}
\begin{definition}[作用量]
    作用量$S$是拉氏量密度在时空上的积分
    \begin{equation}
        S = \int L \, \mathrm{d}t=\int\mathcal L\,\mathrm d^4x
    \end{equation}
\end{definition}
\theorem[标量的Euler-Lagrange方程]
\begin{equation}
    \frac{\partial\mathcal L}{\partial\phi}-\partial_\mu\frac{\partial\mathcal L}{\partial\phi_\mu}=0
\end{equation}
\begin{proof}
    设$\phi\to\phi+\delta\phi$，则
    \begin{equation}
        \delta S = \int \mathrm d^4x \left(\frac{\partial\mathcal L}{\partial\phi}\delta\phi + \frac{\partial\mathcal L}{\partial\phi_\mu}\delta\phi_\mu\right)
    \end{equation}
    对第二项分部积分，忽略边界项，有
    \begin{equation}
        \delta S = \int \mathrm d^4x \left(\frac{\partial\mathcal L}{\partial\phi} - \partial_\mu\frac{\partial\mathcal L}{\partial\phi_\mu}\right)\delta\phi
    \end{equation}
    由$\delta S=0$可得Euler-Lagrange方程
\end{proof}
\begin{example}
    自由粒子的拉氏量密度
    \begin{equation}
        \mathcal L = \frac12\partial_\mu\phi\partial^\mu\phi - \frac12 m^2\phi^2
    \end{equation}
    代入Euler-Lagrange方程, 有
    \begin{equation}
        (\partial_\mu\partial^\mu + m^2)\phi = 0
    \end{equation}
    即Klein-Gordon方程
\end{example}
\begin{theorem}[矢量场的Euler-Lagrange方程]
    设矢量场为$A_\mu$, 
    定义$F_{\mu\nu}=\mathrm dA_{\mu\nu}$, $\Pi^{\mu\nu}=\frac{\partial\mathcal L}{\partial F_{\mu\nu}}, B^\nu=\frac{\partial \mathcal L}{\partial A_\nu}$, 那么
    \begin{equation}
        B^\mu-2\partial_\nu\Pi^{[\nu\mu]}=0
    \end{equation}
\end{theorem}
\begin{proof}
    \begin{equation}
        \delta\mathcal L=\Pi^{\mu\nu}\delta \mathrm dA_{\mu\nu}+\frac{\partial\mathcal L}{\partial A_\mu}\delta A_\mu
    \end{equation}
    交换$\delta$与外微分算子$\mathrm d$, 并利用乘法法则有:
    \begin{subequations}
        \begin{align}
            \delta\mathcal L&=2\partial_{[\mu}(\Pi^{\mu\nu}\delta A_{\nu]})-2(\partial_{[\mu}\Pi^{\mu\nu})\delta A_\nu+\frac{\partial\mathcal L}{\partial A_\mu}\delta A_\mu\\
            &=2\partial_{[\mu}(\Pi^{\mu\nu}\delta A_{\nu]})-2(\partial_\nu\Pi^{[\nu\mu]})\delta A_\mu+\frac{\partial\mathcal L}{\partial A_\mu}\delta A_\mu
        \end{align}
    \end{subequations}
    忽略边界项, 由$\delta S=0$可得Euler-Lagrange方程.
\end{proof}
\begin{example}
    电磁场的拉氏量密度
    \begin{equation}
        \mathcal L = -\frac14 F_{\mu\nu}F^{\mu\nu}-J^\mu A_\mu
    \end{equation}
    代入Euler-Lagrange方程, 有
    \begin{equation}
        \partial_\mu F^{\mu\nu} = J^\nu
    \end{equation}
    即Maxwell方程.
\end{example}
\subsection{Noether定理}
\label{sec:classical-noether}
\songti
本节按照从变分对称性到守恒流、再到规范恒等式的顺序，整理经典场论中的Noether定理。基本思路来自笔记\cite{LinkSymmetry2026}，但沿用本书的$(+---)$度规和标量场正动能约定。特别要区分三个概念：运动方程的对称性、作用量的变分对称性，以及一般协变性。第一Noether定理与第二Noether定理都从\textbf{离壳的变分对称性}出发；两者的区别在于变换参数是否为任意的局域函数。

\subsubsection{对称性、拉回与变分}
若一个变换将运动方程的解映射为另一个解，就称它是运动方程的对称性。Noether定理使用一个更强的充分条件：变换后的作用量与原作用量至多相差一个边界项，而且这一关系对任意场配置成立，不必预先使用运动方程。满足这一条件的变换称为\textbf{变分对称性}。因此，不能仅由“变换保持运动方程”就直接套用下面的守恒流公式。

为使时空变换和体积测度的变化都能被明确追踪，先在定向的$d$维时空$(M,g)$上讨论。仍用$\mathcal L$表示本书原有的标量拉氏量，以粗体$\boldsymbol{\mathcal L}$表示相应的顶次微分形式：
\begin{equation}
    \boldsymbol{\mathcal L}=\mathcal L\,\epsilon,
    \qquad S=\int_M\boldsymbol{\mathcal L},
    \qquad \epsilon=\sqrt{|\det g|}\,\d^dx.
\end{equation}
在四维平直时空的惯性坐标中，这就还原为$S=\int\d^4x\,\mathcal L$。下文的希腊指标在一般$d$维讨论中取$0,\ldots,d-1$，$i$只在$\phi^i$等记号中用作场的种类指标；无穷小参数记为$s$，与体积形式$\epsilon$区分。

考虑连续变换$\psi_s:(M,g,\phi^i)\to(M_s,g'_s,\phi'_s{}^i)$，且$\psi_0$为恒等映射。先将变换后的量拉回原流形，再比较它们：
\begin{align}
    \widehat g_s&=\psi_s^*g'_s,
    &\widehat\phi_s^i&=\psi_s^*\phi'_s{}^i,\\
    \Delta g^{\mu\nu}&=\left.\frac{\d\widehat g_s^{\mu\nu}}{\d s}\right|_{s=0},
    &\Delta\phi^i&=\left.\frac{\d\widehat\phi_s^i}{\d s}\right|_{s=0}.
\end{align}
这里$\widehat g_s^{\mu\nu}$是拉回度规的逆，$\Delta$表示一阶变化的系数。使用正向拉回时，标量的纯时空变换给出$\Delta\phi=\mathscr L_\xi\phi=\xi^\mu\nabla_\mu\phi$。

如果拉氏量由度规、场及其协变导数等几何量构成，并且所有背景结构都一同变换，则有微分同胚协变性
\begin{equation}
    \psi^*\boldsymbol{\mathcal L}[g',\phi']
    =\boldsymbol{\mathcal L}[\psi^*g',\psi^*\phi'].
\end{equation}
对于保定向的$\psi$，体积形式也满足$\psi^*\epsilon[g']=\epsilon[\psi^*g']$。这不是任意坐标表达式都自动具有的性质；显式指定的外场、坐标依赖或方向也必须纳入背景结构。更不能由一般协变性推出任意微分同胚都是\emph{固定背景}上的物理对称性：还要检查拉回后的背景是否仍为原背景。

本节采用的无穷小变分对称性条件是
\begin{equation}
    \boxed{\Delta\boldsymbol{\mathcal L}
    =\d(\iota_K\epsilon)
    =(\nabla_\mu K^\mu)\epsilon.}
    \label{eq:noether-variational-symmetry}
\end{equation}
其中$\iota_K$表示向量$K$对形式的内缩；也可将$\iota_K\epsilon$写作一形式$K_\mu\d x^\mu$的Hodge对偶。式\eqref{eq:noether-variational-symmetry}须\textbf{离壳成立}。在紧支撑变分或适当的边界条件下，这个全微分不改变体内的运动方程。以下均采用这种边界条件，不讨论大变换或非平凡边界荷。

\subsubsection{第一Noether定理}
先考虑不含曲率或度规导数的一阶标量场理论，
\begin{equation}
    \mathcal L=\mathcal L(g^{\mu\nu},\phi^i,\nabla_\mu\phi^i),
    \qquad \Pi_i^\mu:=\frac{\partial\mathcal L}{\partial(\nabla_\mu\phi^i)}.
\end{equation}
由于$\phi^i$是标量，$\Delta(\nabla_\mu\phi^i)=\nabla_\mu\Delta\phi^i$。逆度规变分引起的体积形式变化为
\begin{equation}
    \Delta\epsilon=-\frac12g_{\mu\nu}\Delta g^{\mu\nu}\epsilon.
\end{equation}
因此，变分时既要改变$\mathcal L$，也要改变$\epsilon$：
\begin{align}
    \Delta\boldsymbol{\mathcal L}
    =\bigg[&\frac{\partial\mathcal L}{\partial\phi^i}\Delta\phi^i
    +\Pi_i^\mu\nabla_\mu\Delta\phi^i\notag\\
    &+\left(\frac{\partial\mathcal L}{\partial g^{\mu\nu}}
    -\frac12g_{\mu\nu}\mathcal L\right)\Delta g^{\mu\nu}\bigg]\epsilon.
\end{align}
按体积形式定义Euler--Lagrange导数$E_i$，并采用与本书正能量约定一致的Hilbert能动张量：
\begin{align}
    \delta_\phi S&=\int_M E_i\,\delta\phi^i\epsilon
    \quad\text{（紧支撑变分）},\\
    E_i:=\frac{\delta S}{\delta\phi^i}
    &=\frac{\partial\mathcal L}{\partial\phi^i}-\nabla_\mu\Pi_i^\mu,\\
    \delta_g S&=\frac12\int_M T_{\mu\nu}\,\delta g^{\mu\nu}\epsilon,
    \qquad T_{\mu\nu}=2\frac{\partial\mathcal L}{\partial g^{\mu\nu}}-g_{\mu\nu}\mathcal L.
    \label{eq:noether-hilbert-t}
\end{align}
最后一个代数表达式限于当前不含度规导数的理论；泛函变分定义则更一般。这里对\emph{逆度规}变分前的系数为正号，使用其他度规或能动张量定义的文献时应成套比较约定。

对场变分的导数项使用乘法法则，得到基本变分恒等式
\begin{equation}
    \boxed{\Delta\boldsymbol{\mathcal L}
    =\left[E_i\Delta\phi^i
    +\nabla_\mu(\Pi_i^\mu\Delta\phi^i)
    +\frac12T_{\mu\nu}\Delta g^{\mu\nu}\right]\epsilon.}
    \label{eq:noether-variation}
\end{equation}
与式\eqref{eq:noether-variational-symmetry}比较，定义流
\begin{equation}
    \boxed{j^\mu=\Pi_i^\mu\Delta\phi^i-K^\mu,}
    \label{eq:noether-current}
\end{equation}
便得到\textbf{离壳恒等式}
\begin{equation}
    \boxed{\nabla_\mu j^\mu
    =-E_i\Delta\phi^i-\frac12T_{\mu\nu}\Delta g^{\mu\nu}.}
    \label{eq:noether-offshell}
\end{equation}

\begin{theorem}[第一Noether定理]
    若一参数连续变换在固定背景上满足离壳变分对称性条件，亦即$\Delta g^{\mu\nu}=0$且式\eqref{eq:noether-variational-symmetry}成立，则式\eqref{eq:noether-current}定义的流在物质场满足运动方程$E_i=0$时守恒：
    \begin{equation}
        \nabla_\mu j^\mu=0.
    \end{equation}
    若背景度规也变化，则还须使$T_{\mu\nu}\Delta g^{\mu\nu}=0$，才能由物质场在壳推出同一守恒律。
\end{theorem}
这里“离壳”描述对称性条件与变分恒等式，“在壳”描述使用运动方程后的守恒结论。如果度规也是动力学变量，应将它的Euler--Lagrange项包括在完整理论中，不能只令物质场满足运动方程。

在四维平直时空中，若空间无穷远处没有流量，则守恒流给出守恒荷
\begin{equation}
    Q=\int\d^3x\,j^0,
    \qquad \frac{\d Q}{\d t}=-\oint_{\infty}\vec j\cdot\d\vec S=0.
\end{equation}
流本身并不唯一。在平直时空中，加入$\partial_\nu B^{[\nu\mu]}$不改变其散度；在相应的边界通量消失时，也不改变守恒荷。这类改进项在能动张量和共形对称性中会再次出现。

\begin{example}[复标量场的$U(1)$对称性]\label{csU1}
    沿用本书的复标量场记号$\psi$，考虑
    \begin{equation}
        \mathcal L=\partial^\mu\psi^*\partial_\mu\psi-m^2\psi^*\psi.
    \end{equation}
    整体相位旋转为$\psi\to\e^{is}\psi$、$\psi^*\to\e^{-is}\psi^*$，其中$s$是常数。动能项和质量项中的相位分别抵消，且度规不变，故可取$K^\mu=0$。将$\psi,\psi^*$视为独立变分变量，有
    \begin{equation}
        \Delta\psi=i\psi,\quad\Delta\psi^*=-i\psi^*,
        \qquad\Pi_\psi^\mu=\partial^\mu\psi^*,\quad
        \Pi_{\psi^*}^\mu=\partial^\mu\psi.
    \end{equation}
    代入第一定理，得到
    \begin{equation}
        \boxed{j^\mu=i(\psi\partial^\mu\psi^*-
        \psi^*\partial^\mu\psi).}
        \label{complex-scalar-u1-current}
    \end{equation}
    直接求散度并代入$(\Box+m^2)\psi=0$及其共轭式，可验证
    \begin{equation}
        \partial_\mu j^\mu
        =i(\psi\Box\psi^*-\psi^*\Box\psi)=0.
    \end{equation}
    质量项换成仅依赖$\psi^*\psi$的势能时，相位对称性与流的表达式仍然成立。
\end{example}

时空变换中的$K^\mu$也可以直接求出。对保持背景度规的Killing向量$\xi$，由协变性和Cartan公式，
\begin{align}
    \Delta\boldsymbol{\mathcal L}
    &=\mathscr L_\xi\boldsymbol{\mathcal L}
    =\d(\iota_\xi\boldsymbol{\mathcal L})
    +\iota_\xi\d\boldsymbol{\mathcal L}
    =\d(\iota_\xi\boldsymbol{\mathcal L}),\\
    K^\mu&=\xi^\mu\mathcal L.
    \label{eq:noether-spacetime-k}
\end{align}
这里$\d\boldsymbol{\mathcal L}=0$是因为它已是顶次形式。Killing条件使$\Delta g^{\mu\nu}=0$，从而这确实是固定背景上的变分对称性。对于严格保持拉氏量的内禀变换可以取$K=0$，但一般的内禀变分对称性也可能有非零边界项，仍应实际检查。

\begin{example}[标量场的时空平移对称性]\label{ex:ct_scalar_translation}
    在平直时空取常向量$\xi^\mu$，并采用正向拉回
    \begin{equation}
        \widehat\phi_s(x)=\phi(x+s\xi),
        \qquad\Delta\phi=\xi^\nu\partial_\nu\phi.
    \end{equation}
    若$\mathcal L=\frac12\partial_\mu\phi\partial^\mu\phi-V(\phi)$，则$\Pi^\mu=\partial^\mu\phi$。因此
    \begin{align}
        j^\mu&=\Pi^\mu\xi^\nu\partial_\nu\phi-\xi^\mu\mathcal L
        =T^{\mu\nu}\xi_\nu,\\
        T_{\mu\nu}&=\partial_\mu\phi\partial_\nu\phi-g_{\mu\nu}\mathcal L.
    \end{align}
    这个例子中正则能动张量与Hilbert能动张量一致。四个独立平移分别给出
    \begin{equation}
        \partial_\mu T^{\mu\nu}=0,
        \qquad P^\nu=\int\d^3x\,T^{0\nu}.
    \end{equation}
    也可以直接验证$\partial_\mu T^{\mu\nu}=(\Box\phi+V'(\phi))\partial^\nu\phi=0$。其中$P^0$是能量，$P^i$是动量。曲时空中若只有某个Killing向量，直接得到的是对应的$T^{\mu\nu}\xi_\nu$守恒流，不能把它理解为存在四个全局平移。
\end{example}

\begin{example}[旋转与Lorentz变换]
    对标量场，固定两个不同的指标$\rho,\sigma$，取
    \begin{equation}
        \xi^\mu=x^\rho\eta^{\mu\sigma}-x^\sigma\eta^{\mu\rho}
        \qquad\text{（$\rho,\sigma$固定，不求和）}.
    \end{equation}
    这也是Killing向量，因此
    \begin{equation}
        M^{\mu\,\rho\sigma}=T^{\mu\nu}\xi_\nu
        =x^\rho T^{\mu\sigma}-x^\sigma T^{\mu\rho}.
    \end{equation}
    利用$T^{\mu\nu}$的对称性和在壳守恒性，
    \begin{equation}
        \partial_\mu M^{\mu\,\rho\sigma}
        =T^{\rho\sigma}-T^{\sigma\rho}
        +x^\rho\partial_\mu T^{\mu\sigma}
        -x^\sigma\partial_\mu T^{\mu\rho}=0.
    \end{equation}
    两个指标均为空间方向时，这是角动量流；其中一个为时间方向时，得到boost对应的流。对于具有自旋的场，使用正则能动张量构造Lorentz流还须包括场的自旋部分。
\end{example}
\begin{example}[自由电磁场的时空平移对称性]
    在平直时空中取
    \begin{align}
        \mathcal L=-\frac14F_{\mu\nu}F^{\mu\nu},
        \qquad F_{\mu\nu}=\partial_\mu A_\nu-\partial_\nu A_\mu.
    \end{align}
    对常平移向量$\xi^\nu$, 场的变分为$\Delta A_\rho=\xi^\nu\partial_\nu A_\rho$, 且$K^\mu=\xi^\mu\mathcal L$. 将$A_\rho$的各分量视为场变量, 有
    \begin{align}
        \Pi^{\mu\rho}:=\frac{\partial\mathcal L}{\partial(\partial_\mu A_\rho)}=-F^{\mu\rho},
        \qquad j^\mu=\xi^\nu T^\mu{}_{\nu,\mathrm{can}},
    \end{align}
    其中正则能动张量为
    \begin{align}
        T^\mu{}_{\nu,\mathrm{can}}
        =-F^{\mu\rho}\partial_\nu A_\rho-\delta^\mu{}_{\nu}\mathcal L.
    \end{align}
    这个表达式一般不对称，也不是规范不变量。使用自由场方程$\partial_\rho F^{\rho\mu}=0$，添加改进项后可得到对称且规范不变的Hilbert能动张量：
    \begin{align}
        T^{\mu\nu}
        &=-F^{\mu\rho}F^\nu{}_{\rho}
          +\frac14\eta^{\mu\nu}F_{\rho\sigma}F^{\rho\sigma},\\
        T^{\mu\nu}-T_{\mathrm{can}}^{\mu\nu}
        &=F^{\mu\rho}\partial_\rho A^\nu
          \overset{\mathrm{EOM}}{=}\partial_\rho(F^{\mu\rho}A^\nu).
    \end{align}
    改进项中$F^{\mu\rho}A^\nu$对$\mu,\rho$反对称, 所以其双重散度恒为零. 两个张量均在壳守恒, 且在相应表面项消失时给出相同的总能量和动量. 特别地,
    \begin{align}
        T^{00}=\frac12(\boldsymbol E^2+\boldsymbol B^2).
    \end{align}
    这里还应区分Lie导数与一般的分量微分: 对微分形式, Cartan公式$\mathscr L_\xi=\mathrm d\iota_\xi+\iota_\xi\mathrm d$给出
    \begin{align}
        [\mathscr L_\xi,\mathrm d]=0,
        \qquad \mathscr L_\xi F=\mathrm d(\mathscr L_\xi A).
    \end{align}
\end{example}

\begin{example}[Dirac场的时空平移对称性]
    先考虑平直时空中的常平移. 此时不需要引入旋量在一般微分同胚下的变换, 只需取
    \begin{align}
        \mathcal L=\bar\psi(i\gamma^\mu\partial_\mu-m)\psi,
        \qquad \Delta\psi=\xi^\nu\partial_\nu\psi,
        \qquad \Delta\bar\psi=\xi^\nu\partial_\nu\bar\psi.
    \end{align}
    将$\psi$与$\bar\psi$作为独立的变分变量, 并保持旋量乘积的书写次序. 对此拉氏量, 含场变分导数的边界项为$i\bar\psi\gamma^\mu\Delta\psi$, 而$K^\mu=\xi^\mu\mathcal L$, 因此
    \begin{align}
        j^\mu=\xi^\nu T^\mu{}_{\nu,\mathrm{can}},
        \qquad
        T^\mu{}_{\nu,\mathrm{can}}
        =i\bar\psi\gamma^\mu\partial_\nu\psi
        -\delta^\mu{}_{\nu}\mathcal L.
    \end{align}
    利用Dirac方程及其共轭方程, 可得$\partial_\mu T^\mu{}_{\nu,\mathrm{can}}=0$. 正则张量一般不对称; 对Lorentz变换, 还需计入旋量本身的变换以及相应的自旋流. 若要从几何变分得到对称的Hilbert能动张量, 应引入标架和自旋联络, 不能直接套用前面一阶标量场的度规变分公式.
\end{example}

\begin{definition}[共形变换]
    设$(M,g)$与$(M',g')$为两个带度规的流形。若微分同胚$\psi:M\to M'$满足
    \begin{equation}
        \psi^*g'=\Omega^2g,\qquad\Omega>0,
    \end{equation}
    则称$\psi$为共形变换，其中$\Omega$是$M$上的光滑函数。
\end{definition}
Weyl变换与共形等距变换是这一定义的两种特例\cite{LinkConformalZhihu}：
\begin{itemize}
    \item \textbf{Weyl变换：}取$M'=M$、$\psi=\mathrm{id}_M$，于是$g'=\Omega^2g$。流形上的点保持不变，度规作局域尺度变换。
    \item \textbf{共形等距变换：}取$M'=M$、$g'=g$，于是$\psi^*g=\Omega^2g$。它是同一背景$(M,g)$上的微分同胚，在局部坐标中可写为$x'^\mu(p)=x^\mu(\psi(p))$。
\end{itemize}
下面先讨论Weyl对称性对能动张量迹的约束，再给出共形Killing向量及其守恒流。

\begin{example}[Weyl变换与迹恒等式]
Weyl变换保持流形上的点不变，改变背景度规及场的局域尺度：
\begin{equation}
    g_{\mu\nu}\to\Omega^2g_{\mu\nu},
    \qquad\phi^i\to\Omega^{-w_i}\phi^i,
    \qquad\Omega(x)>0.
\end{equation}
这里$w_i$是场的Weyl权重。设$\Omega=\e^{s\sigma(x)}$，取一阶变化，
\begin{equation}
    \delta_\sigma g^{\mu\nu}=-2\sigma g^{\mu\nu},
    \qquad\delta_\sigma\phi^i=-w_i\sigma\phi^i.
\end{equation}
对完整作用量作变分并分部积分，使用式\eqref{eq:noether-hilbert-t}的泛函定义，
\begin{align}
    \delta S&=\int_M\left[E_i\delta\phi^i
    +\frac12T_{\mu\nu}\delta g^{\mu\nu}\right]\epsilon
    +\text{边界项},\\
    \delta_\sigma S&=-\int_M\sigma
    \left(T^\mu{}_{\mu}+\sum_iw_i\phi^iE_i\right)\epsilon
    +\text{边界项}.
\end{align}
这个变分公式允许曲率耦合，此时$T_{\mu\nu}$包含对曲率变分产生的项。若理论具有Weyl对称性，且$\sigma(x)$紧支撑于时空内部，则边界项没有贡献，$\delta_\sigma S=0$对任意$\sigma(x)$离壳成立。因此
\begin{equation}
    \boxed{T^\mu{}_{\mu}=-\sum_iw_i\phi^iE_i.}
    \label{eq:noether-weyl-identity}
\end{equation}
最后才令物质场满足$E_i=0$，得到在壳无迹条件$T^\mu{}_{\mu}=0$。它也解释了式\eqref{eq:noether-offshell}中的度规项何时消失：若$\Delta g^{\mu\nu}=-2\sigma g^{\mu\nu}$，则该项正比于迹。
\end{example}

\begin{example}[共形耦合的实标量场]
    在$d>2$维中取
    \begin{equation}
        w=\frac{d-2}{2},\qquad
        \xi_c=\frac{d-2}{4(d-1)},\qquad
        \mathcal L=\frac12\nabla_\mu\phi\nabla^\mu\phi
        +\frac12\xi_cR\phi^2.
    \end{equation}
    不难证明，经过Weyl变换$g'_{\mu\nu}=\Omega^2g_{\mu\nu}$、
    $\phi'=\Omega^{-w}\phi$，拉氏量仅相差一个全导数项：
    \begin{equation}
        \boldsymbol{\mathcal L}'-\boldsymbol{\mathcal L}
        =-\frac{w}{2}\nabla_\mu\!\left(\phi^2\nabla^\mu\ln\Omega\right)\epsilon.
    \end{equation}
    这里$\boldsymbol{\mathcal L}=\mathcal L\epsilon$，右端各量均按变换前的场和度规定义。
    因此在边界项消失时，作用量不变。
    取曲率约定$[\nabla_\mu,\nabla_\nu]v^\rho
    =R^\rho{}_{\sigma\mu\nu}v^\sigma$、$R_{\mu\nu}=R^\rho{}_{\mu\rho\nu}$，
    对作用量作度规变分，得到Hilbert能动张量
    \begin{align}
        T_{\mu\nu}
        &=\nabla_\mu\phi\nabla_\nu\phi
        -\frac12g_{\mu\nu}(\nabla\phi)^2\notag\\
        &\quad+\xi_c\left[G_{\mu\nu}\phi^2
        +(g_{\mu\nu}\Box-\nabla_\mu\nabla_\nu)\phi^2\right],
    \end{align}
    其中$G_{\mu\nu}=R_{\mu\nu}-\frac12g_{\mu\nu}R$，
    $\Box=\nabla_\mu\nabla^\mu$。利用
    $\Box\phi^2=2(\nabla\phi)^2+2\phi\Box\phi$，取迹得
    \begin{align}
        T^\mu{}_\mu
        &=-\frac{d-2}{2}(\nabla\phi)^2
        +\xi_c\left[-\frac{d-2}{2}R\phi^2+(d-1)\Box\phi^2\right]\notag\\
        &=\frac{d-2}{2}\phi(\Box\phi-\xi_cR\phi).
    \end{align}
    代入场方程$-\Box\phi+\xi_cR\phi=0$，便有$T^\mu{}_\mu=0$，
    因而这个Hilbert能动张量确实是在壳无迹的。
\end{example}

\begin{example}[共形等距变换与其守恒流]
共形等距变换的生成元$\xi$称为共形Killing向量，满足

\subsubsection{第二Noether定理}

Noether第一定理从有限个常参数的变分对称性得到在壳守恒律；第二定理则讨论参数可以是任意局域函数的规范对称性。两者都要求变分对称性对任意场构型成立，区别并不是一个在壳、另一个离壳。以下规范参数均取紧支撑，以排除边界贡献。

设规范变换的无穷小形式为
\begin{equation}
    \delta_\lambda\phi^i=R^i{}_A\lambda^A,
\end{equation}
其中$R^i{}_A$是作用于参数$\lambda^A(x)$的线性微分算子，其系数可以依赖场及其导数。沿用按体积形式$\epsilon$定义的Euler--Lagrange表达式$E_i$，作用量的变分为
\begin{equation}
    \delta S=\int_M E_i\,\delta\phi^i\,\epsilon+\text{边界项}.
\end{equation}
通过分部积分定义$R^i{}_A$的形式伴随：
\begin{equation}
    \int_M E_i\,(R^i{}_A\lambda^A)\,\epsilon
    =\int_M\lambda^A(R^i{}_A)^\dagger E_i\,\epsilon.
\end{equation}
这里的配对是变分中的双线性配对，$\dagger$表示把参数上的导数移到另一侧，\emph{不对复系数取复共轭}；复场及其共轭场按独立变量变分。

\begin{theorem}[第二Noether定理]
    若作用量对上述任意紧支撑参数$\lambda^A(x)$的变换离壳不变，则Euler--Lagrange表达式满足离壳恒等式
    \begin{equation}
        \boxed{(R^i{}_A)^\dagger E_i\equiv0.}
        \label{eq:noether-second-identity}
    \end{equation}
\end{theorem}
\begin{proof}
    由规范不变性和形式伴随的定义，
    \begin{equation}
        0=\delta_\lambda S
        =\int_M\lambda^A(R^i{}_A)^\dagger E_i\,\epsilon.
    \end{equation}
    参数$\lambda^A(x)$可以任意选取，故其系数逐点为零。整个推导没有令$E_i=0$。
\end{proof}

例如，若变换至多含参数的一阶导数，
\begin{equation}
    \delta_\lambda\phi^i
    =r^i{}_A\lambda^A+r^{i\mu}{}_A\nabla_{\mu}\lambda^A,
\end{equation}
则第二定理具体给出
\begin{equation}
    \boxed{r^i{}_AE_i-\nabla_{\mu}\bigl(r^{i\mu}{}_AE_i\bigr)\equiv0.}
\end{equation}
这说明各个运动方程之间存在微分恒等关系，而不是再使用运动方程证明一个守恒流。若参数带有时空指标，分部积分时也须对这些指标使用相应的协变导数。

\begin{example}[Einstein引力与收缩Bianchi恒等式]
    取度规$g_{\mu\nu}$为动力学变量，考虑纯Einstein--Hilbert作用量
    \begin{equation}
        S[g]=\frac1{2\kappa}\int_MR\,\epsilon,
        \qquad \kappa=8\pi G_{\mathrm N}.
    \end{equation}
    沿用前面的曲率约定，并记Einstein张量为
    \begin{equation}
        G_{\mu\nu}=R_{\mu\nu}-\frac12g_{\mu\nu}R.
    \end{equation}
    任意紧支撑向量场$\xi^{\mu}(x)$生成微分同胚变换
    \begin{equation}
        \delta_\xi g_{\mu\nu}=\mathscr L_\xi g_{\mu\nu}
        =\nabla_{\mu}\xi_{\nu}+\nabla_{\nu}\xi_{\mu}.
    \end{equation}
    顶次形式拉氏量的变化是全微分：
    \begin{equation}
        \delta_\xi\boldsymbol{\mathcal L}
        =\frac1{2\kappa}\nabla_{\mu}(R\xi^{\mu})\,\epsilon,
        \qquad \delta_\xi S=0.
    \end{equation}
    另一方面，对协变度规$g_{\mu\nu}$作变分，略去边界项后有
    \begin{equation}
        E_g^{\mu\nu}:=\frac{\delta S}{\delta g_{\mu\nu}}
        =-\frac1{2\kappa}G^{\mu\nu}.
    \end{equation}
    因而
    \begin{equation}
        0=\delta_\xi S
        =\int_M2E_g^{\mu\nu}\nabla_{\mu}\xi_{\nu}\,\epsilon
        =-\int_M\xi_{\nu}\,2\nabla_{\mu}E_g^{\mu\nu}\,\epsilon.
    \end{equation}
    利用$\xi_{\nu}(x)$的任意性，得到
    \begin{equation}
        \boxed{-2\nabla_{\mu}E_g^{\mu\nu}
        =\frac1\kappa\nabla_{\mu}G^{\mu\nu}\equiv0.}
    \end{equation}
    这就是收缩Bianchi恒等式；它对任意度规成立。
\end{example}

\begin{example}[Maxwell场的规范恒等式]
    在固定背景度规上取
    \begin{equation}
        \mathcal L=-\frac14F_{\mu\nu}F^{\mu\nu},
        \qquad F_{\mu\nu}=\nabla_{\mu}A_{\nu}-\nabla_{\nu}A_{\mu}.
    \end{equation}
    局域规范变换为$\delta_\lambda A_{\mu}=\nabla_{\mu}\lambda$。由于$\lambda$是标量，
    \begin{equation}
        \delta_\lambda F_{\mu\nu}
        =\nabla_{\mu}\nabla_{\nu}\lambda-\nabla_{\nu}\nabla_{\mu}\lambda=0,
    \end{equation}
    故作用量离壳不变。对$A_{\mu}$变分给出
    \begin{equation}
        E_A^{\mu}:=\frac{\delta S}{\delta A_{\mu}}=\nabla_{\nu}F^{\nu\mu}.
    \end{equation}
    对任意紧支撑$\lambda(x)$，
    \begin{equation}
        0=\delta_\lambda S
        =\int_M E_A^{\mu}\nabla_{\mu}\lambda\,\epsilon
        =-\int_M\lambda\nabla_{\mu}E_A^{\mu}\,\epsilon,
    \end{equation}
    因此第二定理成为
    \begin{equation}
        \boxed{-\nabla_{\mu}E_A^{\mu}=-\nabla_{\mu}\nabla_{\nu}F^{\nu\mu}\equiv0.}
    \end{equation}
    平直时空中，这由偏导数的对易性与$F^{\mu\nu}$的反对称性直接得到；在一般无挠背景上，双重散度可写成$\tfrac12[\nabla_{\mu},\nabla_{\nu}]F^{\nu\mu}$，产生的Ricci张量缩并也因对称性而为零。这一恒等式不需要使用Maxwell方程。
\end{example}

\begin{example}[标量QED：相互关联的运动方程]
    沿用本笔记的$(+---)$号差，取
    \begin{equation}
        \begin{aligned}
            \mathcal L&=-\frac14F_{\mu\nu}F^{\mu\nu}
            +(D_{\mu}\phi)^*D^{\mu}\phi-V(\rho),\\
            D_{\mu}\phi&=(\nabla_{\mu}-ieA_{\mu})\phi,
            \qquad \rho=\phi^*\phi.
        \end{aligned}
    \end{equation}
    局域$U(1)$变换为
    \begin{equation}
        A_{\mu}'=A_{\mu}+\nabla_{\mu}\lambda,
        \qquad \phi'=e^{ie\lambda}\phi,
        \qquad \phi^{*'}=e^{-ie\lambda}\phi^*.
    \end{equation}
    因为$F'_{\mu\nu}=F_{\mu\nu}$、$D_{\mu}'\phi'=e^{ie\lambda}D_{\mu}\phi$以及$\rho'=\rho$，拉氏量严格不变。

    将$\phi,\phi^*$视作独立变量，并记$V'=\mathrm dV/\mathrm d\rho$，得到
    \begin{align}
        E_A^{\mu}&=\nabla_{\nu}F^{\nu\mu}
        +ie\bigl[\phi^*D^{\mu}\phi-\phi(D^{\mu}\phi)^*\bigr],\\
        E_\phi&=-(D_{\mu}D^{\mu}\phi)^*-V'\phi^*,\\
        E_{\phi^*}&=-D_{\mu}D^{\mu}\phi-V'\phi.
    \end{align}
    无穷小变换为$\delta A_{\mu}=\nabla_{\mu}\lambda$、$\delta\phi=ie\lambda\phi$和$\delta\phi^*=-ie\lambda\phi^*$。分部积分后，
    \begin{equation}
        0=\delta_\lambda S
        =\int_M\lambda\bigl[-\nabla_{\mu}E_A^{\mu}
        +ie\phi E_\phi-ie\phi^*E_{\phi^*}\bigr]\,\epsilon.
    \end{equation}
    因此离壳恒等式为
    \begin{equation}
        \boxed{-\nabla_{\mu}E_A^{\mu}+ie\phi E_\phi-ie\phi^*E_{\phi^*}\equiv0.}
    \end{equation}
    直接代入也能验证：势能项互相抵消，$F^{\mu\nu}$的双重散度恒为零，其余各项利用
    \begin{equation}
        \nabla_{\mu}\bigl[\phi^*D^{\mu}\phi-\phi(D^{\mu}\phi)^*\bigr]
        =\phi^*D_{\mu}D^{\mu}\phi-\phi(D_{\mu}D^{\mu}\phi)^*
    \end{equation}
    恰好抵消。整个验证没有使用任何一个运动方程；正是规范对称性使这些运动方程并非相互独立。
\end{example}
\subsubsection{Noether's trick}
这个所谓的trick在很多地方都有误解，并且还错误地被称为Noether第二定理。这个说法是完全不对的，Noether's Trick其实与Noether第一定理是等价的：它从离壳成立的全局变分对称性出发，得到在EOM意义下成立的守恒律。将它误解为Noether第二定理大概是因为它也暂时将全局的参数给变为了局域参数，但这不影响它的物理实质。

先沿用前面不含曲率或度规导数的一阶标量场理论。所谓Noether's Trick，它的标准操作就是，我们在做对称性操作的时候，临时将变换参数$s$提升为局域的$s(x)$，取$\delta\phi^i=s(x)\Delta\phi^i$、$\delta g^{\mu\nu}=s(x)\Delta g^{\mu\nu}$。这样Lagrangian的变化就会在我们之前的基础上再多一项，变成
\begin{equation}
    \begin{aligned}
        \delta_s\boldsymbol{\mathcal L}
        &=\left[\frac{\delta S}{\delta\phi^i}\Delta\phi^i
        +\nabla_\mu(\Pi_i^\mu\Delta\phi^i)
        +\frac12T_{\mu\nu}\Delta g^{\mu\nu}\right]s\epsilon\\
        &\quad+(\nabla_\mu s)\Pi_i^\mu\Delta\phi^i\epsilon.
    \end{aligned}
\end{equation}
然后前面这一项由离壳的变分对称性条件，可以变为$s\d{*K}$。这一步本身不需要$T_{\mu\nu}\Delta g^{\mu\nu}=0$；这个条件会在下面利用EOM令$\delta S=0$时用到。于是
\begin{align}
    \delta_s\boldsymbol{\mathcal L}
    &=\left[s\nabla_\mu K^\mu
    +(\nabla_\mu s)\Pi_i^\mu\Delta\phi^i\right]\epsilon\notag\\
    &=\d({*sK})+(\nabla_\mu s)(\Pi_i^\mu\Delta\phi^i-K^\mu)\epsilon.
\end{align}
我们通过对$s$施以紧支撑条件，可以去掉全部的边界项，因此我们有
\begin{equation}
    \delta S=\int(\nabla_\mu s)(\Pi_i^\mu\Delta\phi^i-K^\mu)\epsilon.
    \label{eq:noether-trick-standard}
\end{equation}
然后再次分部积分，利用$s$的紧支撑条件，仍然可以去掉边界项。在物质场满足EOM、并且$T_{\mu\nu}\Delta g^{\mu\nu}=0$时（特别是固定背景$\Delta g^{\mu\nu}=0$时），有$\delta S=0$，因此
\begin{equation}
    \delta S=-\int s\nabla_\mu(\Pi_i^\mu\Delta\phi^i-K^\mu)\epsilon=0.
    \label{noether-trick}
\end{equation}
因为$s(x)$是任意的，如果我们想要对任意的$s$都满足$\delta S=0$，只能要求
\begin{equation}
    \nabla_\mu(\Pi_i^\mu\Delta\phi^i-K^\mu)=0.
\end{equation}
从而我们同样证明了守恒流。因此我们发现，如果在计算中施加一个local的$s$控制的变换，$\delta S$中正比于$\nabla_\mu s$的这一项恰为我们想要求的守恒流。

然后我们接下来再以共形变换为例介绍如何使用Noether's Trick计算守恒流。

\paragraph{用Noether's Trick再求共形流。}
现在考虑微分同胚协变且Weyl不变的完整作用量，允许包含曲率耦合。将式\eqref{eq:noether-fixed-conformal}局域化，取固定背景上的场变分
\begin{equation}
    \delta_s\phi^i=s(x)\bigl(\xi^\mu\nabla_\mu\phi^i+w_i\sigma\phi^i\bigr),
    \qquad\delta_s g^{\mu\nu}=0.
\end{equation}
为计算作用量的变化，辅助考虑由$v=s\xi$生成的微分同胚，再叠加参数为$-s\sigma$的Weyl变换。这个联合变换对场的作用恰好等于上式，但逆度规变化为
\begin{align}
    \delta_{\mathrm{comb}}g^{\mu\nu}
    &=-\nabla^\mu(s\xi^\nu)-\nabla^\nu(s\xi^\mu)
    +2s\sigma g^{\mu\nu}\notag\\
    &=-\xi^\nu\nabla^\mu s-\xi^\mu\nabla^\nu s.
\end{align}
第二步使用共形Killing方程。这里不能丢掉$\nabla s$项：$\xi$是共形Killing向量，$s(x)\xi$一般不是。

取$s$紧支撑，联合变换由微分同胚协变性和Weyl对称性保证作用量离壳不变。它与所求固定背景变换之间仅差上述度规变分，因此
\begin{align}
    0=\delta_{\mathrm{comb}}S
    &=\delta_sS+\frac12\int_M T_{\mu\nu}
    \delta_{\mathrm{comb}}g^{\mu\nu}\epsilon\notag\\
    &=\delta_sS-\int_M(\nabla_\mu s)T^{\mu\nu}\xi_\nu\epsilon.
\end{align}
于是直接得到
\begin{equation}
    \boxed{\delta_sS=\int_M(\nabla_\mu s)T^{\mu\nu}\xi_\nu\epsilon,
    \qquad J_\xi^\mu=T^{\mu\nu}\xi_\nu.}
\end{equation}
再利用物质场在壳时的驻值条件，便有$\nabla_\mu J_\xi^\mu=0$。整个推导使用完整作用量的Hilbert变分，因而包含曲率耦合带来的改进项，也与前面给出的共形流一致。

\subsection{哈密顿量}
\begin{definition}[共轭动量]
    共轭动量
    \begin{equation}
        \Pi = \frac{\partial\mathcal L}{\partial\dot\phi}
    \end{equation}
\end{definition}
做Legendre变换, 有Hamiltonian:
\begin{equation}
    H(\phi, \Pi, \nabla\phi) = \int \mathrm d^3x \, \Pi\dot\phi-L=\int \mathrm d^3x \, (\Pi\dot\phi-\mathcal L),
\end{equation}
定义Hamiltonian密度:
\begin{equation}
    \mathcal H(\phi, \Pi, \nabla\phi) = \Pi\dot\phi - \mathcal L,
\end{equation}
定义Poisson括号$\{, \}: \mathcal F\times\mathcal F\to\mathbb{R}$:
\begin{equation}
    \{F, G\}=\displaystyle\int \mathrm d^3x \left(\frac{\delta F}{\delta\phi(x)}\frac{\delta G}{\delta\Pi(y)}-\frac{\delta G}{\delta\phi(y)}\frac{\delta F}{\delta\Pi(x)}\right)
\end{equation}
\begin{example}
    \begin{equation}
        \{\phi(\mathbf x), \Pi(\mathbf y)\} = \delta^3(\mathbf x - \mathbf y)
    \end{equation}
\end{example}
\begin{proof}
    设$F=\phi(\mathbf x), G=\Pi(\mathbf y)$, 则
    \begin{equation}
        \frac{\delta F}{\delta\phi(x)} = \delta^3(\mathbf x - \mathbf x'), \quad \frac{\delta F}{\delta\Pi(x)} = 0
    \end{equation}
    \begin{equation}
        \frac{\delta G}{\delta\phi(x)} = 0, \quad \frac{\delta G}{\delta\Pi(x)} = \delta^3(\mathbf y - \mathbf x')
    \end{equation}
    代入Poisson括号定义, 有
    \begin{equation}
        \{\phi(\mathbf x), \Pi(\mathbf y)\} = \int \mathrm d^3x' (\delta^3(\mathbf x - \mathbf x')\delta^3(\mathbf y - \mathbf x') - 0) = \delta^3(\mathbf x - \mathbf y)
    \end{equation}
\end{proof}
\theorem[哈密顿正则方程]
\begin{equation}
    \begin{cases}
        \dot\phi(\mathbf x) = \{\phi_{\mathbf x}, H\}=\frac{\delta H}{\delta\Pi}=\frac{\partial\mathcal H}{\partial\Pi}\\
        \dot\Pi(\mathbf x) = \{\Pi_{\mathbf x}, H\}=-\frac{\delta H}{\delta\phi}=-\frac{\partial\mathcal H}{\partial\phi}+\partial_i\frac{\partial\mathcal H}{\partial(\phi_i)}
    \end{cases}
\end{equation}
\begin{proof}
    \begin{equation}
        \begin{split}
            \delta H &= \int \mathrm d^3x \left(\delta\Pi_\mathbf x\dot\phi_\mathbf x+\Pi_\mathbf x\delta\dot\phi_\mathbf x-\frac{\partial\mathcal L}{\partial\phi}\delta\phi-\Pi_\mathbf x\delta\dot\phi-\frac{\partial\mathcal L}{\partial\phi_i}\delta\phi_i\right)\\
            &= \int \mathrm d^3x \left(\delta\Pi_\mathbf x\dot\phi_\mathbf x - \frac{\partial\mathcal L}{\partial\phi}\delta\phi - \frac{\partial\mathcal L}{\partial\phi_i}\delta\phi_i\right)\\
        \end{split}
    \end{equation}
    分部积分并消去边缘项有:
    \begin{equation}
        \delta H= \int \mathrm d^3x \left(\delta\Pi_\mathbf x\dot\phi_\mathbf x - \frac{\partial\mathcal L}{\partial\phi}\delta\phi + \partial_i\frac{\partial\mathcal L}{\partial\phi_i}\delta\phi\right)\\
    \end{equation}
    于是可得:
    \begin{equation}
        \begin{cases}
            \dot\phi(\mathbf x) = \frac{\partial\mathcal H}{\partial\Pi}\\
            \dot\Pi(\mathbf x) = -\frac{\partial\mathcal H}{\partial\phi}+\partial_i\frac{\partial\mathcal H}{\partial(\phi_i)}
        \end{cases}
    \end{equation}
\end{proof}
讨论: 在转换到Hamilton力学的过程中, Legendre变换给予了$\dot\phi$特殊的地位, 使得我们选定的参考系的时间轴$t$具有了特殊的地位, 因此破坏了洛伦兹协变性.

\section{二次量子化}
\songti 二次量子化是将场(如电磁场、电子场)本身进行量子化的框架，它将描述单粒子概率幅的经典场提升为场算符，其激发则对应粒子的产生与湮灭，从而自然地描述了粒子数可变的多粒子系统. 相比之下, 一次量子化中粒子是给定的，其运动（波函数）是量子的.
\subsection{自由实标量场的量子化\label{2nd_quant_free_real_scalar}}
% \begin{enumerate}
%     \item 将场的动力学方程转换为算符方程
%     \item 找到动力学方程的一般解
%     \item 将一般解的积分常数升级为常算符
%     \item 施加量子化条件
%     \item 用常算符构造Hilbert空间
% \end{enumerate}
\begin{enumerate}
    \item 写下Lagrangian
    \item 得到Hamiltonian
    \item 做正则变换解耦
    \item 施加量子化条件
    \item 得到产生湮灭算符
    \item 得到场方程
\end{enumerate}
\note{这里流程没按周洋讲的来, 因为我略微感觉他那样子做有一点点奇怪, 有些地方存在神秘的天降系数, 按照这样从Lagrangian出发的流程会更清晰一些.}

写下Lagrangian:
\begin{equation}
    \mathcal L=\frac12\partial_\mu\phi\partial^\mu\phi - \frac12 m^2\phi^2,
\end{equation}
我们有EoM:
\begin{equation}
    (\Box+m^2)\phi = 0,
\end{equation}
从而可以得到色散关系:
\begin{equation}
    \omega^2=\vec p^2+m^2,
\end{equation}
满足这个色散关系的, 或者满足EoM的粒子称为称为on-shell的粒子.

我们可以计算出正则动量
\begin{align}
    \pi=\pa{\mathcal L}{\dot\phi}=\dot\phi,
\end{align}
于是我们可以得到哈密顿量
\begin{equation}
    \mathcal H=\frac12(\pi^2+m^2\phi^2+(\nabla\phi)^2).
\end{equation}

使用母函数
\begin{equation}
    U=-\int\Pi(\vec k)\pi(\vec x)\mathrm e^{i\vec k\cdot\vec x}\mathrm d^3x\mathrm d^3k,
\end{equation}
进行正则变换可以得到
\begin{align}
    &\phi(\vec x)=\int\Phi_{\vec k}\mathrm e^{i\vec k\cdot\vec x}\mathrm d^3k,\\
    &\pi(\vec x)=\frac1{(2\pi)^3}\int\Pi_{\vec k}\mathrm e^{-i\vec k\cdot\vec x}\mathrm d^3k.
\end{align}
\begin{align}
    &\Phi_{\vec k}=\frac1{(2\pi)^3}\int\phi(\vec x)\mathrm e^{-i\vec k\cdot\vec x}\mathrm d^3x,\\
    &\Pi_{\vec k}=\int\pi(\vec x)\mathrm e^{i\vec k\cdot\vec x}\mathrm d^3x.
\end{align}
并且有:
\begin{equation}
    \Phi_{\vec k}^\dagger=\Phi_{-\vec k}, \quad \Pi_{\vec k}^\dagger=\Pi_{-\vec k},
\end{equation}
可以得到解耦后的Hamiltonian:
\begin{equation}
    H=\frac12\int \mathrm d^3k (\frac1{(2\pi)^3}\Pi_{\vec k}\Pi_{-\vec k}+(2\pi)^3\omega^2\Phi_{\vec k}\Phi_{-\vec k}),
\end{equation}
其中$\omega^2=m^2+k^2$.

添加量子化条件:
\begin{equation}
    [\phi(\vec x, t),\pi(\vec y, t)]=i\delta^3(x-y),
\end{equation}
于是有:
\begin{equation}
    [\Phi_{\vec k}, \Pi_{\vec p}]=i\delta^3(k-p).
\end{equation}
令
\begin{align}
    &a_{\vec k}=(2\pi)^3\sqrt{\frac{\omega}{2}}\left(\Phi_{\vec k}+\frac i{\omega}\frac1{(2\pi)^3}\Pi_{\vec k}^\dagger\right)=\frac1{\sqrt{2\om k}}\int\d^3x\exp{ikx}(\omega\phi+i\pi),\\
    &a^\dagger_{\vec k}=(2\pi)^3\sqrt{\frac{\omega}{2}}\left(\Phi_{\vec k}^\dagger-\frac i{\omega}\frac1{(2\pi)^3}\Pi_{\vec k}\right)=\frac1{\sqrt{2\om k}}\int\d^3x\exp{-ikx}(\omega\phi-i\pi),
\end{align}
于是有对易子:
\begin{equation}
    [a_{\vec k}, a_{\vec p}^\dagger]=(2\pi)^3\delta^3(\vec k-\vec p).
\end{equation}
更准确地说, 对于$[a_{\vec k}, a_{\vec k}^\dagger]$, 
\begin{equation}
    [a_{\vec k}, a_{\vec k}^\dagger]=\mathcal V, 
\end{equation}
$\mathcal V$为系统的总体积.

于是Hamiltonian可以被对角化:
\begin{equation}
    H=\int\frac{\mathrm d^3p}{(2\pi)^3}\omega_{\vec p}(a^\dagger_{\vec p} a_{\vec p}+\frac12\mathcal V),
\end{equation}

Fourier逆变换可以得到:
\begin{align}
    &\phi(\vec x)=\int\frac{\mathrm d^3p}{(2\pi)^3}\frac1{\sqrt{2\omega_{\vec p}}}(a^\dagger_{\vec p}\mathrm e^{-i\vec p\cdot\vec x}+a_{\vec p}\mathrm e^{i\vec p\cdot\vec x})\\
    &\pi(\vec x)=\int\frac{\mathrm d^3p}{(2\pi)^3} i\sqrt{\frac{\omega_{\vec p}}{2}}(a^\dagger_{\vec p} e^{-i\vec p\cdot\vec x}-a_{\vec p} e^{i\vec p\cdot\vec x}),
\end{align}

最后利用时间演化算符得到场方程:
\begin{important}
    \begin{equation}\label{ch4freephi}
        \phi(x)=\mathrm e^{iHt}\phi(\vec x)\mathrm e^{-iHt}=\int\frac{\mathrm d^3p}{(2\pi)^3}\frac1{\sqrt{2\omega_{\vec p}}}(a_{\vec p}^\dagger\mathrm e^{ipx}+a_{\vec p}\mathrm e^{-ipx})
    \end{equation}
    \begin{equation}\label{ch4freepi}
        \pi(x)=\mathrm e^{iHt}\pi(\vec x)\mathrm e^{-iHt}=\int\frac{\mathrm d^3p}{(2\pi)^3} i\sqrt{\frac{\omega_{\vec p}}{2}}(a^\dagger_{\vec p} \e^{ipx}-a_{\vec p} \e^{-ipx})
    \end{equation}
\end{important}

我们可以计算所谓的真空零点能密度:
\begin{equation}
    \frac{\bra0 H\ket0}{\mathcal V}=\int\frac{\mathrm d^3p}{(2\pi)^3}\frac{\omega_{\vec p}}{2}=\int\frac{\mathrm d^3p}{(2\pi)^3}\frac{\sqrt{p^2+m^2}}{2}=+\infty,
\end{equation}

我们还可以定义动量算子:
\begin{equation}
    P^i=\int\partial^0\phi\,\partial^i\phi\,\mathrm d^3x=-\int\phi_t\phi_i\mathrm d^3x,
\end{equation}
即:
\begin{align}
    \vec P&=-\int\phi_t\nabla\phi\mathrm d^3x\\
    &=-\int\mathrm d^3x\int\ldsq{p}[i\omega_{\vec p}](a^\dagger_{\vec p} e^{ipx}-a_{\vec p} e^{-ipx})\\
    &\quad\int\ldsq{q}[i\vec q](-a^\dagger_{\vec q}\exp{iqx}+a_{\vec q}\exp{-iqx})\\
    &=\int\ld{p}[\vec p\omega_{\vec p}](a^\dagger_{\vec p} a_{\vec p}+a_{\vec p}a^\dagger_{\vec p})\\
    &=\int\ddd p\vec p a^\dagger_{\vec p} a_{\vec p},
\end{align}

这里需要补充一点:
\begin{theorem}[Lorentz不变的体元]
    \begin{equation}
        \int\frac{\mathrm d^3p}{(2\pi)^3}\frac1{2E}
    \end{equation}
    是一个Lorentz不变量
\end{theorem}
\begin{proof}
    考虑
    \begin{equation}
        \int\frac{\mathrm d^4p}{(2\pi)^4}\Theta(p^0) 2\pi\delta(p^2-m^2)
    \end{equation}
    可以发现其等于
    \begin{equation}
        \int\frac{\mathrm d^3p}{(2\pi)^3}\frac1{2E}
    \end{equation}
\end{proof}
然后我们就可以定义真空态以及Fock空间:
\begin{definition}[真空态]
    真空态$\ket0$满足
    \begin{equation}
        a_{\vec p}\ket0=0, \quad \forall \vec p
    \end{equation}
\end{definition}
\begin{definition}[Fock空间]
    Fock空间为
    \begin{equation}
        \mathcal H=\bigoplus_{n=1}\mathcal H_n
    \end{equation}
    其中$\mathcal H_n$为$n$粒子空间, 即:
    \begin{equation}
        \mathcal H_n=\mathrm{span}\{a_{\vec p_1}^\dagger a_{\vec p_2}^\dagger \cdots a_{\vec p_n}^\dagger \ket0 \,|\, \forall \vec p_i\}
    \end{equation}
\end{definition}
接下来我们检查自由标量场的二次量子化结果与我们的经典一次量子化结果相一致.

\begin{definition}[动量本征态]
    \begin{equation}
        \ket{\vec p}=\sqrt{2\omega_{\vec p}}a^\dagger_{\vec p}\ket0
    \end{equation}
    \begin{equation}
        \ket{\vec p\vec q}=\sqrt{4\omega_{\vec p}\omega_{\vec q}}a^\dagger_{\vec p}a^\dagger_{\vec q}\ket0
    \end{equation}
\end{definition}
我们不难验证:
\begin{equation}
    \braket{\vec p|\vec q}=2\omega_{\vec p}(2\pi)^3\delta^3(\vec p-\vec q),
\end{equation}
\begin{equation}
    \braket{\vec p'\vec q'|\vec p\vec q}=4\omega_{\vec p}\omega_{\vec q}(2\pi)^6\left(\delta^3(\vec p-\vec p')\delta^3(\vec q-\vec q')+\delta^3(\vec p'-\vec q)\delta^3(\vec q'-\vec p)\right),
\end{equation}
\begin{equation}
    \vec P\ket{\vec p}=\vec p\ket{\vec p}
\end{equation}

\begin{definition}[位置本征态]
    定义位置产生算符:
    \begin{align}
        &\psi^\dagger(x)=\int\ldsq pa^\dagger_{\vec p}\mathrm e^{ipx}\\
        &\psi(x)=\int\ldsq pa_{\vec p}\mathrm e^{-ipx}
    \end{align}
    以及位置本征态
    \begin{equation}
        \ket{\vec x}=\psi^\dagger(x)\ket0
    \end{equation}
\end{definition}
我们可以发现:
\begin{equation}
    \ket{\vec x}=\psi^\dagger(x)\ket0=\int\ld{p}\exp{ipx}\ket{\vec p},
\end{equation}
\begin{equation}
    \braket{\vec p|\vec x}=\exp{ipx},
\end{equation}

可以发现与我们在QM里学的一致.

\begin{example}[实标量自由场的真空波函数]
    考虑质量$m>0$的自由实标量场, 在$t=0$的场本征态表象中, 有
    \begin{equation}
        \hat\phi(\vec x)\ket\phi=\phi(\vec x)\ket\phi,
        \qquad \Psi[\phi]=\braket{\phi|\Psi}.
    \end{equation}
    这里$\ket\phi$由整个空间上的场构型$\phi(\vec x)$标记, 因而$\Psi[\phi]$是一个波泛函. 与QM中的坐标表象完全类似, 场算符与共轭动量算符分别表示为
    \begin{equation}
        \hat\phi(\vec x)\longrightarrow\phi(\vec x),
        \qquad \hat\pi(\vec x)\longrightarrow-i\frac{\delta}{\delta\phi(\vec x)}.
    \end{equation}
    于是前面定义的湮灭算符在此表象中为($\omega_{\vec k}=\sqrt{\vec k^2+m^2}$)
    \begin{align}
        a_{\vec k}
        &=\frac1{\sqrt{2\omega_{\vec k}}}\int\d^3x\,\e^{-i\vec k\cdot\vec x}
        \left(\omega_{\vec k}\hat\phi(\vec x)+i\hat\pi(\vec x)\right)\notag\\
        &\longrightarrow\frac1{\sqrt{2\omega_{\vec k}}}\int\d^3x\,\e^{-i\vec k\cdot\vec x}
        \left(\omega_{\vec k}\phi(\vec x)+\frac{\delta}{\delta\phi(\vec x)}\right).
    \end{align}
    记真空波泛函为$\Psi_0[\phi]=\braket{\phi|0}$, 并定义
    \begin{equation}
        \widetilde\phi(\vec k)=\int\d^3x\,\e^{-i\vec k\cdot\vec x}\phi(\vec x).
    \end{equation}
    由$a_{\vec k}\ket0=0$, 得到
    \begin{equation}
        \int\d^3x\,\e^{-i\vec k\cdot\vec x}
        \frac{\delta\Psi_0[\phi]}{\delta\phi(\vec x)}
        =-\omega_{\vec k}\widetilde\phi(\vec k)\Psi_0[\phi].
    \end{equation}
    对其做Fourier逆变换, 即
    \begin{align}
        \frac{\delta\Psi_0[\phi]}{\delta\phi(\vec x)}
        &=-\int\d^3y\,K(\vec x-\vec y)\phi(\vec y)\Psi_0[\phi],\notag\\
        K(\vec x-\vec y)&=\int\frac{\d^3k}{(2\pi)^3}\,
        \omega_{\vec k}\e^{i\vec k\cdot(\vec x-\vec y)}.
    \end{align}
    利用$K(\vec x-\vec y)=K(\vec y-\vec x)$, 可以直接解得
    \begin{equation}
        \boxed{\Psi_0[\phi]=\mathcal N\exp{
        -\frac12\int\d^3x\,\d^3y\,\phi(\vec x)K(\vec x-\vec y)\phi(\vec y)
        }},
    \end{equation}
    或者写成动量空间的形式
    \begin{equation}
        \boxed{\Psi_0[\phi]=\mathcal N\exp{
        -\frac12\int\frac{\d^3k}{(2\pi)^3}\,\omega_{\vec k}
        \widetilde\phi(\vec k)\widetilde\phi(-\vec k)}}.
    \end{equation}
    其中$\mathcal N$为归一化常数; 实场满足$\widetilde\phi(-\vec k)=\widetilde\phi(\vec k)^*$, 此时真空期望值$\bra0\hat\phi(\vec x)\ket0=0$.

    可以看到，自由场具有和简谐振子完全一样的特性：基态的场值在真空期望值附近是一个gaussian的涨落.
\end{example}

\subsection{自由复标量场量子化}
\begin{equation}
    \mathcal L=\partial_\mu\psi^*\partial^\mu\psi-m^2\psi^*\psi,
\end{equation}
其中$\psi=\phi_1+i\phi_2$

\begin{align}
    \pi^\mu=\partial^\mu\psi^*,
\end{align}
\begin{align}
    \pi=\pi^0=\partial^0\psi^*=\partial_0\psi^*,
\end{align}

正则变换:
\begin{align}
    \begin{cases}
        &\Psi_{\vec k}=\frac1{\dpi3}\int\d^3x\psi_{\vec x}\exp{-i\vec k\cdot\vec x}\\
        &\Pi_{\vec k}=\int\d^3x\pi_{\vec x}\exp{i\vec k\cdot\vec x}
    \end{cases},
\end{align}

于是得到:
\begin{equation}
    H=\int\d^3k\left(\frac1{\dpi3}\Pi^\dagger_{-\vec k}\Pi_{\vec k}+\dpi3\om p^2\Psi_{-\vec k}^\dagger\Psi_\vec k\right),
\end{equation}

添加正则量子化条件:
\begin{equation}
    [\psi_{\vec x}, \pi_{\vec y}]=i\delta^3(\vec x-\vec y), \quad [\psi_\vec x, \psi_\vec y]=[\pi_\vec x, \pi_\vec y]=0,
\end{equation}

于是我们得到:
\begin{align}
    [\Psi_\vec p, \Pi_\vec q]=[\Psi^\dagger_\vec p, \Pi^\dagger_\vec q]=i\delta^3(\vec p-\vec q), \quad [\Psi_\vec p, \Psi_\vec q]=[\Pi_\vec p, \Pi_\vec q]=0,
\end{align}

设
\begin{align}
    &\a p=\dpi3\sqrt{\frac{\om p}2}(\Psi_\vec p+\frac i{\dpi3\om p}\Pi^\dagger_{-\vec p})=\frac1{\sqrt{2\om p}}\int\d^3x\exp{ipx}(\om p\psi+i\pi^\dagger)\\
    &\b p=\dpi3\sqrt{\frac{\om p}2}(\Psi_{-\vec p}^\dagger+\frac i{\dpi3\om p}\Pi_{\vec p})=\frac1{\sqrt{2\om p}}\int\d^3x\exp{-ipx}(\om p\psi^\dagger+i\pi),
\end{align}

我们有:
\begin{align}
    [\a p, \a q^\dagger]&=\dpi3\delta^3(\vec p-\vec q), \quad [\a p, \a q]=[\a p^\dagger, \a q^\dagger]=0\\
    [\b p, \b q^\dagger]&=\dpi3\delta^3(\vec p-\vec q), \quad [\b p, \b q]=[\b p^\dagger, \b q^\dagger]=0,
\end{align}

然后可以对角化Hamiltonian
\begin{equation}
    H=\int\ddd p\om p\left(\a p^\dagger\a p+\b p^\dagger\b p+\mathcal V\right),
\end{equation}


然后有:
\begin{important}
    \begin{align}
        &\psi(x)=\int\ldsq p\left(\a p\exp{-ipx}+\b p^\dagger\exp{ipx}\right)\\
        &\pi(x)=\int\ddd pi\sqrt{\frac{\om p}2}\left(\a p^\dagger\exp{ipx}-\b p\exp{-ipx}\right)
    \end{align}
\end{important}

例\ref{csU1}指出复标量场的$U(1)$对称性导致有守恒荷$Q$
\begin{equation}
    Q\equiv\int d^3x J_N^0=\int d^3xi\left[(\partial^0\psi^\dagger)\psi-(\partial^0\psi)\psi^\dagger\right],
\end{equation}

经过计算我们可以得到:
\begin{equation}
    Q_N=\int\ddd p\left(\b p^\dagger\b p-\a p^\dagger\a p\right),
\end{equation}

并且有对易关系
\begin{align}\label{complex-scalar-Q-commutation-relations}
    &[Q_N, \psi_x]=\psi_x\\
    &[Q_N, \psi_x^\dagger]=-\psi_x^\dagger,
\end{align}

\note[粒子数与守恒荷]{复标量场是双自由度的, 因此它有两个产生湮灭算子$a, b$. $a, b$分别代表正粒子和反粒子的运动模式. 而从其的$U(1)$对称性导出的守恒量中可以得出结论, 正粒子与反粒子数目之差为常数.}

\subsection{两点关联函数}
简单地说, 关联函数就是时空中$n$个不同的点处的场乘积的平均:
$$
    \braket{\phi_1\cdots\phi_n}.
$$
这个定义可能比较唐突, 我们难以理解为什么我们需要在场论中计算关联函数, 为什么关联函数非常重要. 因此在具体计算之前我们想给出一些具体的关联函数的意义的例子, 说明它的重要性和物理意义.

我们知道, 某一个态可以通过场算符激发出来
\begin{align}
    \ket{X}=\phi(x)\cdots\ket\Omega,
\end{align}
经过演化后一个态到另一个态的概率振幅为
\begin{align}
    \braket{Y|X}=\braket{\Omega|\phi(y)\cdots\phi(x)|\Omega},
\end{align}
某个可观测量测量的平均值为
\begin{align}
    \braket{O}=\braket{X|O|X}=\braket{\Omega|\cdots\phi(x)O\phi(x)\cdots|\Omega},
\end{align}
可以看到, 这就是一个关联函数的形式. 在此基础之上, 因为, 微分散射截面、衰变率甚至能流分布等物理量本质上就是通过这两个表达式来表达, 因此也可以通过关联函数算出来.

这是我们常见的粒子物理中的理解, 但是其可应用的范围远远不止于此. 在腾飞的讲义中给出了两个更加有意思的例子\cite{gft:lecture1}: 其一是黑洞散射发射出的引力波波形的计算, 可以表达为
$$
    h_{\mu\nu}(\tau, \theta, \phi)=\braket{\rm{in}|S^\dagger h_{\mu\nu} S|\rm{in}}_{\text{classical limit}}
$$
也就是说, 就连经典场的散射我们原则上也是可以利用QFT关联函数计算的框架最终取经典极限$\hbar\to0$得到! 

再一个则是宇宙微波背景辐射的关联函数: 微波背景辐射近似是各向同性的, 这一点可以通过暴涨模型很好地解释(宇宙尺度很大, 但是可以让其中的光子气处于热平衡态, 只能是在宇宙很小的时候宇宙两点间相互作用很强的时代就已经达成了平衡). 但是实际上更精细的实验指出, 它实际上是存在微小涨落的, 而这个涨落的平均正是关联函数, 并且可以通过弯曲时空QFT计算出来!
$$
    \braket{\Omega|\phi(x_1)\phi(x_2)\phi(x_3)\phi(x_4)|\Omega}
$$

\subsubsection{实标量场}\label{2pt-real-scalar}
\begin{definition}[关联函数$D(x-y)$\cite{peskinCausality}]
    \begin{equation}
        D(x-y)\equiv\braket{0|\phi(x)\phi(y)|0}
    \end{equation}
\end{definition}
计算可得:
\begin{theorem}
    \begin{align}
        D(x-y)&=\int\ldsq{p}\int\ldsq{q}\exp{iqy-ipx}\braket{0|\a p\ad q|0}\\
        &=\int\ld{p}\exp{-ip(x-y)}
    \end{align}
\end{theorem}
可以发现, 这是一个Lorentz不变量(更准确来说, 是$\mathcal P\times\rm{SO}(1,3)$的不变量, 由于存在$\Theta$它不能在$\mathcal T$下不变).

\note[两点关联函数的意义]{不难证明上一节中的$\braket{x|y}$正是两点关联函数, 也就是说, 它表示在时空点$y$处(假设y更早发生)激发一个粒子, 在时空点$x$处测量到它的概率密度(当然我们这里还要模长平方一下). 在凝聚态场论中这个意义可能更加明确, 产生湮灭算符的关联函数模长平方就是在$t$时刻$\vec x$处产生粒子后, 在$t'$时刻$\vec x'$处粒子数密度的变化.}

我们分类讨论类时、类空间隔的关联函数$D(x-y)$:\\
类时: 我们不妨假设$x^0>y^0$, 那么我们可以做Lorentz变换, 使得$\vec x'=\vec y'=0$. 设$t={x'}^0-{y'}^0$, 则:
\begin{equation}
    D(x-y)=\int\ld p\exp{-i\omega_{\vec p}t'}=\int_m^{+\infty}\sqrt{\omega^2-m^2}\exp{-i\omega t'}\frac{\d\omega}{\dpi2},
\end{equation}
对于$t'\rightarrow+\infty$
\begin{equation}
    D(x-y)\sim\exp{-imt'},
\end{equation}
类空: 我们可以做Lorentz变换, 使得$x^0=y^0=0$, 并设$\vec r=\vec x-\vec y$, 则:
\begin{align}
    D(x-y)&=\int\ld p\exp{i\vec p\cdot(\vec x-\vec y)}\\
    &=\int_0^{+\infty}\frac{2\pi p^2\d p}{\dpi3}\frac1{2\omega_{\vec p}}\int_0^\pi\exp{ipr\cos\theta}\sin\theta\d\theta\\
    &=\int_0^{+\infty}\frac{-i}{2\dpi2}\frac{\exp{ipr}-\exp{-ipr}}r\frac{p\d p}{\sqrt{p^2+m^2}}\\
    &=-\frac i{2\dpi2r}\int_{-\infty}^{+\infty}\frac{p}{\sqrt{p^2+m^2}}\exp{ipr}\d p,
\end{align}
注意到, $p=\pm im$是被积分函数的两个支点, 于是将积分路径改为图\ref{fig:q2Dspace}中沿着上半部分支割线的路径. 然后可得:
\begin{equation}
    D(x-y)=\frac1{\dpi2 r}\int_m^{+\infty}\frac{\omega\exp{-\omega r}}{\sqrt{\omega^2-m^2}}\d\omega,
\end{equation}
对于$r\rightarrow+\infty$
\begin{equation}
    D(x-y)\sim\exp{-mr},
\end{equation}
我们发现, 即使是类空间隔, $D(x-y)$仍不为0, 这说明$\phi(x), \phi(y)$在空间中存在重叠(overlap).

\note[类空间隔与因果律]{类空间隔的关联函数非零能说明这违反了因果律吗？并不能, 因为所谓的因果律需要是指类空间隔的测量之间互相不影响, 也就是说交换这两个算子作用在态上的顺序不会影响结果. 这暗示我们或许应当计算两个算符的对易子来检验我们的理论是否违背了因果律, 而最直接的检验就是计算$[\phi(x), \phi(y)]$.}

因此我们考虑$\phi(x), \phi(y)$的对易子:
\begin{equation}
    [\phi(x), \phi(y)]=\braket{0|[\phi(x), \phi(y)]|0}=D(x-y)-D(y-x),
\end{equation}

再次尝试对类时类空间隔分类讨论:\\
类空: 首先将换参考系, 使得$x$, $y$在同一三维空间中, 然后利用Parity算符$\mathcal P$, 显然可以使得$D(x-y)\rightarrow D(y-x)$, 而$\mathcal P$操作不会改变结果, 于是我们有:
\begin{equation}
    [\phi(x), \phi(y)]=0,
\end{equation}
类时: 由于$[\phi(x), \phi(y)]$不在$\mathcal T$中保持不变, 因此我们无法如类空般做到交换$x$, $y$, 因此我们不能得到$[\phi(x), \phi(y)]=0$.

可以发现, 对易子对类空间隔一定为0而对类时间隔则不一定, 这正是我们想要的因果性!

然后我们尝试进一步计算对易子
\begin{align}
    [\phi(x), \phi(y)]&=D(x-y)-D(y-x)\\
    &=\int\ld p(\exp{-ip(x-y)}-\exp{ip(x-y)})\\
    &=\int\ddd p(\frac1{2\omega_p}\exp{-ip(x-y)}+\frac1{-2\omega_p}\exp{ip(x-y)}),
\end{align}
注意到对第二项做$\vec p\rightarrow-\vec p$换元结果不变, 于是:
\begin{align}
    [\phi(x), \phi(y)]&=\int\ddd p\exp{i\vec p\cdot(\vec x-\vec y)}\left(\frac{\exp{-i\omega_p(x^0-y^0)}}{2\omega_p}+\frac{\exp{i\omega_p(x^0-y^0)}}{-2\omega_p}\right),
\end{align}

观察这个形式, 令我们想到留数定理, 这两项就是留数的相加, 我们可以将其化为:
\begin{align}
    \frac{\exp{-i\omega_p(x^0-y^0)}}{2\omega_p}+\frac{\exp{i\omega_p(x^0-y^0)}}{-2\omega_p}&=\frac{1}{-2\pi i}\left(-2\pi i\rm{Res}(...)-2\pi i\rm{Res}(...)\right)\label{q2eq1}\\
    &=\frac i{2\pi}\int_C\d\omega\frac{\exp{-i\omega(x^0-y^0)}}{(\omega+\omega_{\vec p})(\omega-\omega_{\vec p})}\\
    &=\frac i{2\pi}\int_C\d\omega\frac{\exp{-i\omega(x^0-y^0)}}{\omega^2-\vec p^2-m^2}\\
    &=\int_C\frac{\d\omega}{2\pi}\frac{i}{p^2-m^2}\exp{-i\omega(x^0-y^0)}\label{q2eq2}
\end{align}

于是我们可以定义新的关联函数
\begin{theorem}[推迟关联函数$D_R(x-y)$]
    取围道$C$如图\ref{fig:q2f1}
    \begin{equation}
        D_R(x-y)\equiv\int\dddd p \frac i{p^2-m^2}\exp{-ip(x-y)}\label{q2eq3}
    \end{equation}
    或者等价于取一个无穷小正数$\epsilon$:
    \begin{equation}
        D_R(x-y)\equiv\lim_{\epsilon\rightarrow0^+}\int\dddd p \frac i{(p+i\epsilon)^2-m^2}\exp{-ip(x-y)}
    \end{equation}
    (这里$p+i\epsilon$的意思是对$p^0+i\epsilon$, 其三维部分是不动的)
\end{theorem}
不难发现, 对于$x^0<y^0$, 围道内不含有任何pole, 故$D_R(x-y)=0$, 而对于$x^0>y^0$, 围道内含有pole, 故$D_R(x-y)=D(x-y)-D(y-x)$

于是我们发现
\begin{equation}
    D_R(x-y)=\Theta(x^0-y^0)[\phi(x), \phi(y)],
\end{equation}
这令人想起电动力学中我们学过的推迟势, 故而得名.

类似地, 我们可以定义提前关联函数
\begin{theorem}[提前关联函数$D_A(x-y)$]
    \begin{equation}
        D_A(x-y)\equiv\lim_{\epsilon\rightarrow0^+}\int\dddd p \frac i{(p-i\epsilon)^2-m^2}\exp{-ip(x-y)}\label{q2eq4}
    \end{equation}
\end{theorem}
我们可以发现:
\begin{equation}
    D_A(x-y)=-\Theta(y^0-x^0)[\phi(x), \phi(y)],
\end{equation}

这正与电动力学中所谓的提前势相对应.

此外, 我们还有两种取围道的方式, 由此定义两种关联函数, 也就是所谓的Feynman传播子.
% \newpage
\begin{theorem}[Feynman传播子$D_F(x-y)$]
    取围道$C$如图\ref{fig:q2f3}
    \begin{equation}
        D_F(x-y)\equiv\int\dddd p \frac i{p^2-m^2}\exp{-ip(x-y)}\label{q2eq5}
    \end{equation}
    或者等价于取一个无穷小正数$\epsilon$:
    \begin{equation}
        D_F(x-y)\equiv\lim_{\epsilon\rightarrow0^+}\int\dddd p \frac i{p^2-m^2+i\epsilon}\exp{-ip(x-y)}
    \end{equation}
\end{theorem}
\begin{theorem}[共轭Feynman传播子$\widetilde{D}_F(x-y)$]
    % 取围道$C$如图\ref{fig:q2f4}
    % \begin{equation}
    %     \widetilde D_F(x-y)\equiv\int\dddd p \frac i{p^2-m^2}\exp{-ip(x-y)}\label{q2eq6}
    % \end{equation}
    % 或者等价于取一个无穷小正数$\epsilon$:
    \begin{equation}
        \widetilde D_F(x-y)\equiv\lim_{\epsilon\rightarrow0^+}\int\dddd p \frac i{p^2-m^2-i\epsilon}\exp{-ip(x-y)}\label{q2eq6}
    \end{equation}
\end{theorem}

不难发现, 对于$x^0<y^0$或者$x^0>y^0$, 这俩关联函数都存在一个pole, 并且可以计算发现:
\begin{align}
    D_F(x-y)&=\Theta(x^0-y^0)D(x-y)+\Theta(y^0-x^0)D(y-x)\\
    &=\Theta(x^0-y^0)\braket{0|\phi(x)\phi(y)|0}+\Theta(y^0-x^0)\braket{0|\phi(y)\phi(x)|0}\\
    &=\braket{0|\mathcal T\phi(x)\phi(y)|0},
\end{align}
\begin{align}
    \widetilde D_F(x-y)&=-\Theta(x^0-y^0)D(x-y)-\Theta(y^0-x^0)D(y-x)\\
    &=-\Theta(x^0-y^0)\braket{0|\phi(x)\phi(y)|0}-\Theta(y^0-x^0)\braket{0|\phi(y)\phi(x)|0}\\
    &=-\braket{0|\mathcal T\phi(x)\phi(y)|0},
\end{align}
其中$\mathcal T$为时间顺序算符
\begin{definition}[时间顺序算符$\mathcal T$]
    对于一串$\phi(x_1,x_2,\cdots,x_n)$的算符, 时间顺序算符即起到排序作用, 将时间上在后面发生的算符放在左边, 时间上在前面发生的算符放在右边.
\end{definition}

\note[围道的选取]{从\eqref{q2eq1}到\eqref{q2eq2}只有在一定条件下取正确的围道才成立, 并不能认为是一个任何条件下都恒等的关系. 它的作用在于启发性得导出传播子$\frac i{p^2-m^2}$, 并由此根据不同的围道取法定义出四种不同的关联函数. 因此\eqref{q2eq3}、\eqref{q2eq4}、\eqref{q2eq5}、\eqref{q2eq6}并不能视为是从\eqref{q2eq2}直接推导得到的.}

在量子场论中我们只使用Feynman传播子, 这个原因在后面我们有关Dyson级数以及路径积分的讨论中可以清楚地看到. 为了能让我们对于Feynman传播子的讨论对于接下来的工作更有推广借鉴意义, 在此我们给出另一种从关联函数定义
\begin{equation}
    D_F(x, y)=\braket{0|\mathcal T\phi_x\phi_y|0}=\Theta(t_x-t_y)\braket{0|\phi_x\phi_y|0}+\Theta(t_y-t_x)\braket{0|\phi_y\phi_x|0},
\end{equation}
出发来得到Feynman传播子的方式, 这个方法在接下来的推导中更为有用. 

根据式\eqref{Theta-definition}我们有
\begin{align}
    D_F(x, y)=&\int\frac{i\d\omega}{2\pi}\frac{\exp{-i\omega(t_x-t_y)}}{\omega+i\epsilon}\int\ld p[\exp{-ip(x-y)}]\notag\\
    &+\int\frac{i\d\omega}{2\pi}\frac{\exp{i\omega(t_x-t_y)}}{\omega+i\epsilon}\int\ld p[\exp{ip(x-y)}]\label{scalar-feynman-propagator-naively},
\end{align}
(注意, 这里$p(x-y)=\omega(t_x-t_y)-\vec p\cdot(\vec x-\vec y)$). 我们设
\begin{equation}
    p'^\mu=(\omega, \vec p)^T,
\end{equation}
则有
\begin{equation}
    D_F(x, y)=\int i\dddd p'\frac{\exp{-ip'(x-y)-i\om p(t_x-t_y)}}{(p'^0+i\epsilon)2\om p}+\int i\dddd p'\frac{\exp{ip'(x-y)+i\om p(t_x-t_y)}}{(p'^0+i\epsilon)2\om p},
\end{equation}
(注意, 这里$p(x-y)=p^0(t_x-t_y)-\vec p\cdot(\vec x-\vec y)$, 其中$p^0=\omega$, 因此这里$p(x-y)$与式\eqref{scalar-feynman-propagator-naively}的含义是不同的). 再做换元将$\om p$吸收到$p$中
\begin{equation}
    p^0=p'^0+\om p=p'^0+\sqrt{\vec p'^2+m^2},
\end{equation}
这个换元的Jacobi行列式
\begin{equation}
    |\frac{\partial p'}{\partial p}|=1,
\end{equation}
得到
\begin{equation}
    D_F(x, y)=\int i\dddd p\left(\frac{\exp{-ip(x-y)}}{(p^0-\om p+i\epsilon)2\om p}+\frac{\exp{ip(x-y)}}{(p^0-\om p+i\epsilon)2\om p}\right).
\end{equation}
再对第二项做换元, 将$\exp{-ip(x-y)}$提到括号外
\begin{equation}
    p\to -p,
\end{equation}
我们有
\begin{align}
    D_F(x, y)&=\int i\dddd p\frac{\exp{-ip(x-y)}}{2\om p}\left(\frac{1}{p^0-\om p+i\epsilon}+\frac{1}{-p^0-\om p+i\epsilon}\right)\\
    &=\int i\dddd p\exp{-ip(x-y)}\frac1{(p'^0-\om p+i\epsilon)(p'^0+\om p-i\epsilon)}\\
    &=\int i\dddd p\exp{-ip(x-y)}\frac1{(p'^0)^2-\om p^2+2i\om p\epsilon+\epsilon^2},
\end{align}
忽略掉$\epsilon$的二阶项, 并且将$\om p$吸收进$\epsilon$, 利用$\om p^2=\vec p^2+m^2$, 我们有
\begin{align}
    D_F(x, y)&=\int\dddd p\frac i{p^2-m^2+i\epsilon}\exp{-ip(x-y)}.
\end{align}
这就是实标量场的Feynman传播子.

\subsubsection{复标量场}
我们还是首先计算
\begin{align}
    \braket{0|\psi_x\psi_y^\dagger|0}&=\bra0\int\ldsq p\left(\a p^\dagger\exp{ipx}+\b p\exp{-ipx}\right)\notag\\
    &\;\int\ldsq q\left(\a q\exp{-iqy}+\b p^\dagger\exp{iqy}\right)\ket0\\
    &=\int\ld p[\exp{-ip(x-y)}]\\
    \braket{0|\psi_y^\dagger\psi_x|0}&=\bra0\int\ldsq q\left(\a q\exp{-iqy}+\b p^\dagger\exp{iqy}\right)\notag\\
    &\;\int\ldsq p\left(\a p^\dagger\exp{ipx}+\b p\exp{-ipx}\right)\ket0\\
    &=\int\ld p[\exp{ip(x-y)}],
\end{align}

类似地, 我们可以将$\Theta$展开为Fourier积分, 从而给出复标量场的Feynman传播子
\begin{align}
    \braket{0|\mathcal T\psi_x\psi_y^\dagger|0}=\int\dddd p\frac i{p^2-m^2+i\epsilon}\exp{-ip(x-y)},
\end{align}

读者可能会疑问, $\psi_x\psi_y, \psi_x^\dagger\psi_y^\dagger$是否也有传播子? 事实上它们的传播子为$0$, 也就是不传播.
\begin{align}
    \braket{\psi_x\psi_y}=\braket{\psi_x^\dagger\psi_y^\dagger}=0,
\end{align}

这是因为根据$Q\ket0=0$, 即真空态电中性, 我们有
\begin{align}
    \braket{0|\mathcal T\psi_x\psi_y|0}&=\braket{0|\mathcal T\psi_x\psi_y|0}+\braket{0|Q_N\mathcal T\psi_x\psi_y|0}\\
    &=\braket{0|\mathcal T\psi_x\psi_y|0}+\braket{0|\mathcal T\psi_xQ_N\psi_y|0}+\braket{0|\mathcal T\psi_x\psi_y|0}\\
    &=2\braket{0|\mathcal T\psi_x\psi_y|0}+\braket{0|\mathcal T\psi_x\psi_yQ_N|0}+\braket{0|\mathcal T\psi_x\psi_y|0}\\
    &=3\braket{0|\mathcal T\psi_x\psi_y|0},
\end{align}
(需要注意, 因为$Q_N$不含时, 所以可以自由进出$\mathcal T$, 然后利用式\eqref{complex-scalar-Q-commutation-relations}中给出的$Q_N$与$\psi_x$的对易关系). 因此
\begin{align}
    \braket{0|\mathcal T\psi_x\psi_y|0}=0,
\end{align}
同理可得
\begin{align}
    \braket{0|\mathcal T\psi_x^\dagger\psi_y^\dagger|0}=0
\end{align}

\subsubsection{关联函数与Green函数}
\begin{theorem}[关联函数的性质]
    这四个关联函数$D_R(x-y)$, $D_A(x-y)$, $D_F(x-y)$, $\widetilde D_F(x-y)$都是EoM的Green函数, 即:
    \begin{equation}
        \begin{split}
            &(\Box+m^2)D_R(x-y)=-i\delta^4(x-y)\\
            &(\Box+m^2)D_A(x-y)=-i\delta^4(x-y)\\
            &(\Box+m^2)D_F(x-y)=-i\delta^4(x-y)\\
            &(\Box+m^2)\widetilde D_F(x-y)=-i\delta^4(x-y)
        \end{split}
    \end{equation}

    而$D(x-y)$则没有这个性质.
\end{theorem}
\begin{proof}
    记这四种中的某个关联函数为$D_X(x-y)$
    我们从定义出发
    \begin{equation}
        D_X(x-y)=\int\dddd p \frac i{p^2-m^2}\exp{-ip(x-y)}
    \end{equation}

    因此
    \begin{equation}
        \Box D_X(x-y)=\int\dddd p \frac i{p^2-m^2}(-p^2)\exp{-ip(x-y)}
    \end{equation}
    \begin{equation}
        m^2 D_X(x-y)=\int\dddd p \frac i{p^2-m^2}(m^2)\exp{-ip(x-y)}
    \end{equation}
    直接相加, 利用\eqref{matheq1}即得:
    \begin{equation}
        (\Box+m^2)D_X(x-y)=-i\int\dddd p\exp{-ip(x-y)}=-i\delta^4(x-y)
    \end{equation}

    而对于$D(x-y)$
    \begin{align}
        (\Box+m^2)D(x-y)&=\int\ld p(p^2+m^2)\exp{-ip(x-y)}\\
        &=m^2\int\ddd p\frac1{\omega_{\vec p}}\exp{-ip(x-y)}\neq-i\delta^4(x-y)
    \end{align}
\end{proof}
\note[传播子与离壳传播]{这些传播子在经典场论中中对应的就是EoM的Green函数, 这就说明了它们的物理意义正是标量粒子的传播. 而$\dddd p$则暗示着在qft中它的传播是off-shell的, 存在不满足色散关系的所谓"内线"粒子. 反之, 若考虑满足色散关系的实粒子, 则我们需要在$\dddd p$后加上$\Theta(p^0)\delta(p^2-m^2)$, 或者使用三维体元$\ld p$以使其on-shell.}

\subsection{Wick定理}
\begin{definition}[$\mathcal N$算符]
    $\mathcal N$算符可以将输入它的一串$\a{p}$, $\a{p}^\dagger$强行变成$\a{p}^\dagger$在前$\a{p}$在后的顺序. 例如:
    \begin{equation}
        \mathcal N\left\{\a{p}\a{p}^\dagger\a{q}\a{k}^\dagger\right\}=\a{p}^\dagger\a{k}^\dagger\a{q}\a{p}
    \end{equation}
\end{definition}
\begin{definition}[Wick收缩(Wick contract)]
    \begin{equation}
        \overline{\phi(x)\phi(y)}\equiv D_F(x-y)
    \end{equation}
\end{definition}

\begin{theorem}[Wick定理]\label{wicksTheorem}
    \begin{equation}
        \mathcal T\left[\phi(x_1)\phi(x_2)\cdots\phi(x_n)\right]=\mathcal N\left\{\phi(x_1)\phi(x_2)\cdots\phi(x_n)+\text{所有的Wick收缩}\right\}
    \end{equation}
    其中, 所谓的"所有的Wick"收缩指, 任意的一对两两收缩、任意的两对两两收缩、任意的三对两两收缩等等等\\
    比如对于$n=4$
    \begin{equation}
        \begin{split}
            \mathcal T[\phi(x_1)&\phi(x_2)\phi(x_3)\phi(x_4)]
            =\mathcal N\{\phi(x_1)\phi(x_2)\phi(x_3)\phi(x_4)\\
            &+\overline{\phi(x_1)\phi(x_2)}\phi(x_3)\phi(x_4)+\overline{\phi(x_1)\phi(x_3)}\phi(x_2)\phi(x_4)\\
            &+\overline{\phi(x_1)\phi(x_4)}\phi(x_2)\phi(x_3)+\overline{\phi(x_2)\phi(x_3)}\phi(x_1)\phi(x_4)\\
            &+\overline{\phi(x_2)\phi(x_4)}\phi(x_1)\phi(x_3)+\overline{\phi(x_3)\phi(x_4)}\phi(x_1)\phi(x_2)\\
            &+\overline{\phi(x_1)\phi(x_2)}\cdot\overline{\phi(x_3)\phi(x_4)}+\overline{\phi(x_1)\phi(x_3)}\cdot\overline{\phi(x_2)\phi(x_4)}\\
            &+\overline{\phi(x_1)\phi(x_4)}\cdot\overline{\phi(x_2)\phi(x_3)}\}
        \end{split}
    \end{equation}
\end{theorem}
\begin{proof}
    \begin{lemma}\label{lemma:N_commute}
        $\mathcal N$算符可以与$[,]$换序:
        \begin{equation}
            [\a{p}, N(\cdots)]=N([\a{p}, \cdots])
        \end{equation}
    \end{lemma}
    \begin{lemma}\label{lemma:psi_commute}
        对于$x^0>y^0$
        \begin{equation}
            [\psi(x), \phi(y)]=[\psi(x), \psi^\dagger(y)]=D(x-y)=D_F(x-y)
        \end{equation}
    \end{lemma}
    不妨假设$x_1^0\geq x_2^0 \geq x_3^0\geq\cdots x_n^0$
    首先, 对于$n=2$:
    \begin{equation}
        \mathcal T[\phi(x_1)\phi(x_2)]=\mathcal N\{\phi(x_1)\phi(x_2)+\overline{\phi(x_1)\phi(x_2)}\}
    \end{equation}
    显然成立.

    考虑数学归纳法, 若对$n=k-1$成立, 则对$n=k$:
    \begin{align*}
        &\mathcal T\left[\phi(x_1)\phi(x_2)\cdots\phi(x_k)\right]=\phi(x_1)\left[\phi(x_1)\phi(x_2)\cdots\phi(x_k)\right]\\
        &=(\psi(x_1)+\psi^\dagger(x_1))\mathcal N\left\{\phi(x_2)\phi(x_3)\cdots\phi(x_n)+\text{所有的}(k-1)\text{Wick收缩}\right\}\\
        &=\mathcal N\left\{\psi^\dagger(x_1)(\phi(x_2)\phi(x_3)\cdots\phi(x_n)+\text{所有的}(k-1)\text{Wick收缩})\right\}\\
        &~~~~+\mathcal N\left\{(\phi(x_2)\phi(x_3)\cdots\phi(x_n)+\text{所有的}(k-1)\text{Wick收缩})\psi(x_1)\right\}+[\psi(x_1), \mathcal N\{\cdots\}]\\
        &=\mathcal N\left\{\phi(x_1)\phi(x_2)\phi(x_3)\cdots\phi(x_n)+\text{所有的}(k-1)\text{Wick收缩}\right\}+\mathcal N\{[\psi(x_1), \cdots]\}\\
        &=\mathcal N\left\{\phi(x_1)\phi(x_2)\phi(x_3)\cdots\phi(x_n)+\text{所有的}(k-1)\text{Wick收缩}\right\}\\
        &~~~~+\mathcal N\{\text{与}\psi(x_1)\text{收缩后的所有}(k-1)\text{Wick收缩}\}
    \end{align*}

    最终我们合并得到:
    \begin{equation}
        \mathcal T\left[\phi(x_1)\phi(x_2)\cdots\phi(x_k)\right]=\mathcal N\left\{\phi(x_1)\phi(x_2)\phi(x_3)\cdots\phi(x_n)+\text{所有的}k\text{Wick收缩}\right\}
    \end{equation}
    于是根据归纳公理我们证明得到了Wick定理.

    % \note[Wick定理的适用范围]{最后补充一点, 对于Wick定理的证明, 我们并没有利用到任何Free Field的性质, 因此Wick定理也是可以适用于接下来的Interaction Theory的.}
\end{proof}

对于旋量, Wick定理同样满足, 但是在这时因为
\begin{equation}
    \braket{0|\mathcal T\psi_x\bar\psi_y|0}=-\braket{0|\mathcal T\bar\psi_y\psi_x|0}\neq\braket{0|\mathcal T\bar\psi_y\psi_x|0},
\end{equation}
Wick收缩的内顺序(即两个场$\psi, \bar\psi$缩并时谁在前谁在后)是不可忽略的, 不难证明, 正确的缩并方法应该是在缩并内保持$\mathcal T$内的先后顺序.

\subsection{对称性与守恒荷}
考虑某一变换$X$(要求不只是保持$S$不变, 还进一步要求$\mathcal L$不变)
\begin{align}
    \psi_i\to\epsilon\delta_X\psi_i,
\end{align}
我们可以定义出守恒流
\begin{align}
    j^\mu_X=\sum_i\frac{\partial\mathcal L}{\partial(\partial_\mu\psi_i)}\delta_X\psi_i,
\end{align}
并且有守恒荷
\begin{align}
    Q_X&=\int\d^3x j^0_X(x)\\
    &=\int\d^3x\sum_i\frac{\partial\mathcal L}{\partial(\partial_t\psi_i)}\delta_X\psi_i\notag\\
    &=\int\d^3x\sum_i \pi_i(x)\delta_X\psi_i(x).
\end{align}

如果$\delta_X$是commute的, 则我们有如下推导:
对于动量满足对易关系的场,
\begin{align}
    [\psi_i(x), \pi_j(y)]=i\delta_{ij}\delta^3(x-y),
\end{align}
我们有
\begin{align}
    i[Q_X, \psi_i]&=\int\d^3y\sum_j i[\pi_j(y)\delta_X\psi_j(y), \psi_i(x)]\\
    &=\int\d^3y\sum_j i([\pi_j(y), \psi_i(x)]\delta_X\psi_j(y)+\pi_j(y)[\delta_X\psi_j(y), \psi_i(x)]),
\end{align}
若对称性不包含时间上的变换, 则$\delta_X\psi$不含有场对时间的导数, 也就不含正则动量, 因此在等时条件下有
\begin{align}
    [\delta_X\psi_j(y), \psi_i(x)]=0, 
\end{align}
从而
\begin{align}
    i[Q_X, \psi_i]&=\int\d^3y\sum_j i([\pi_j(y), \psi_i(x)]\delta_X\psi_j(y))\\
    &=\int\d^3y\sum_j i(-i\delta_{ij}\delta^3(x-y)\delta_X\psi_j(y))\\
    &=\delta_X\psi_i(x).
\end{align}
对于动量满足反对易关系的场, 我们可以类似推导得到
\begin{align}
    i[Q_X, \psi_i]&=\int\d^3y\sum_j i(\{\pi_j(y), \psi_i(x)\}\delta_X\psi_j(y)-\pi_j(y)\{\delta_X\psi_j(y), \psi_i(x)\})\\
    &=\int\d^3y\sum_j i(-i\delta_{ij}\delta^3(x-y)\delta_X\psi_j(y))\\
    &=\delta_X\psi_i(x).
\end{align}

而对于anti-commute的$\delta_X$, 比如说BRST的, 则有细微的区别, 对于动量位形是对易子关系的场结论仍然不变
\begin{align}
    i[Q_X, \psi_i]=\delta_X\psi_i(x),
\end{align}
但是对于反对易的算子则有
\begin{align}
    i\{Q_X, \psi_i\}=\delta_X\psi_i(x).
\end{align}
值得注意, 这里$Q_X$是anti-commute的.

于是我们有定理
\begin{theorem}\label{conserved-charge-as-generator}
    考虑一保持$\mathcal L$不变的对称性$X$, 场的无穷小变换为$\psi\to\psi+\epsilon\delta_X\psi$, 对于动量与位形为对易关系的算符, 若它满足在等时条件下, 
    \begin{align}
        [\delta_X\psi_j(y), \psi_i(x)]=0,
    \end{align}
    则
    \begin{align}
        i[Q_X, \psi_i]=\delta_X\psi_i(x);
    \end{align}
    而对于动量位形是反对易的算符, 若$\delta_X$是commute的, 仍然有
    \begin{align}
        i[Q_X, \psi_i]=\delta_X\psi_i(x),
    \end{align}
    但是对于anti-commute的$\delta_X$, 则
    \begin{align}
        i\{Q_X, \psi_i\}=\delta_X\psi_i(x).
    \end{align}
\end{theorem}

这告诉我们对于有限大的变换, 我们可以用$Q_X$生成
\begin{align}
    \exp{iQ_X\epsilon}\psi\exp{-iQ_X\epsilon}=X\psi.
\end{align}
这验证了守恒荷确实是这个变换的对应生成元(需要注意, 对于anti-commute的$Q_X$, $\epsilon$也是anti-commute的, 比如BRST对称性就是如此).
